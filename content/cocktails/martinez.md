layout: layouts/cocktail.njk
permalink: /cocktails/{{ slug }}/
ingredients:
  - 1 1/2 oz (4.5 cl) gin
  - 1 1/2 oz (4.5 cl) sweet vermouth
  - 1/6 oz (0.5 cl) maraschino liqueur
  - 2 dashes orange bitters
steps:
  - Stir with ice until chilled.
  - Strain into a coupe.
  - Garnish with cherry.
