layout: layouts/cocktail.njk
permalink: /cocktails/{{ slug }}/
ingredients:
  - 4.5 cl (1 1/2 oz) gin
  - 4.5 cl (1 1/2 oz) sweet vermouth
  - 0.5 cl (1/6 oz) maraschino liqueur
  - 2 dashes orange bitters
steps:
  - Stir with ice until chilled.
  - Strain into a coupe.
  - Garnish with cherry.
